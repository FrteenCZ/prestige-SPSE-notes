# 28b.1 Vhodnými integračními metodami vypočtěte určité integrály:
1) $$
\int_{1}^4(\sqrt{ x }+2x)dx=\left[ \frac{2x^{3/2}}{3} +x^{2}\right]_{1}^4=\frac{59}{3}
$$
2) $$
\begin{align}
\int_{1}^8 \frac{\sqrt{ x }}{\sqrt[3]{ x }}-\frac{x^{2}}{\sqrt[3]{ x }}dx=\int_{1}^8(x^{1/6}-x^{5/3})dx=\left[ \frac{6x^{7/6}}{7}-\frac{3x^{8/3}}{8} \right]_{1}^8
\end{align}
$$
3) $$
\begin{align}
\int_{0}^{\pi/4} \left( \frac{2}{\cos ^{2}x}-1 \right)dx=[-2\cot x-x]_{0}^{\pi/4}
\end{align}
$$
4) $$
\begin{align}
\int_{1}^e\left( -\frac{3}{x}+2 \right)dx=[-3\ln|x|+2x]_{1}^e
\end{align}
$$
5) $$
\begin{align}
\int_{0}^{\pi/2}\cos ^{2}x\sin xdx=\int_{1}^0u^{2}\sin x\frac{du}{-\sin x}=\int_{0}^1u^{2}du=\left[ \frac{u^{3}}{3} \right]_{0}^1=\frac{1}{3}
\end{align}
$$
6) $$
\begin{align}
\int_{0}^1 \frac{x}{(x^{2}+1)^{2}}dx=\int_{1}^{2}u^{-2}x\frac{du}{2x}=\frac{1}{2}\left[- \frac{1}{u} \right]_{1}^2=\frac{1}{4}
\end{align}
$$
7) $$
\begin{align}
\int_{0}^{\pi/2}\frac{\cos x}{1+\sin x}dx=\int_{1}^{2}\frac{1}{u}du=[\ln|u|]_{1}^2=\ln2
\end{align}
$$
8) $$
\begin{align}
\int_{0}^1 x(2x^{2}-1)^{10}dx=\int_{-1}^{1}xu^{10}\frac{du}{4x}=\frac{1}{4}\left[ \frac{u^{11}}{11} \right]_{-1}^{1}=\frac{1}{22}
\end{align}
$$
9) $$
\begin{align}
\int_{0}^ex^{3}\ln xdx=\left[ \frac{x^{4}\ln x}{4}\right]_{0}^e-\int_{0}^e\frac{x^{3}}{4}dx=\frac{1}{4}\left[ x^{4}\ln x\right]_{0}^e-\frac{1}{16}[x^4]_{0}^e
\end{align}
$$
10) $$
\begin{align}
\int_{0}^{\pi/2}x\cos xdx=[x\sin x]_{0}^{\pi/2}-\int_{0}^{\pi/2}\sin xdx=[x\sin x]_{0}^{\pi/2}+[\cos x]_{0}^{\pi/2}=\frac{\pi}{2}-1
\end{align}
$$
