# 8. Jsou dána komplexní čísla $a=-6+5i$, $b=4-7i$, $c=\cos \frac{3}{4}\pi+i\sin \frac{3}{4}\pi$. Sestrojte graficky v Gaussově rovině obraz komplexního čísla $z=\overline{a}-b+2c$. Ověřte početně.

```tikz
\usepackage{amsmath}
\usetikzlibrary{arrows.meta, calc}

\begin{document}
\begin{tikzpicture}[>=Stealth, scale=0.6]
    % Draw Grid
    \draw[very thin, gray!30] (-13,-7) grid (6,6);
    
    % Draw Axes
    \draw[->, thick] (-13,0) -- (6,0) node[right] {$Re$};
    \draw[->, thick] (0,-7) -- (0,6) node[above] {$Im$};
    
    % Define Coordinates
    \coordinate (O) at (0,0);
    \coordinate (A) at (-6,5);
    \coordinate (B) at (4,-7);
    \coordinate (C) at ({-sqrt(2)/2}, {sqrt(2)/2});
    \coordinate (TwoC) at ({-sqrt(2)}, {sqrt(2)});
    
    % The result z = Abar - B + 2C
    % z = (-6-4-1.41) + (-5+7+1.41)i = -11.41 + 3.41i
    \coordinate (Z) at ({-10-sqrt(2)}, {2+sqrt(2)});

    % Draw Vectors
    \draw[->, cyan, thick] (O) -- (A) node[below left] {$a$};
    \draw[->, red, thick] (O) -- (B) node[below right] {$b$};
    \draw[->, orange, thick] (O) -- (TwoC) node[above left] {$2c$};
    
    % Construction lines for z = (Abar - B) + 2C
    % First, vector -b
    \coordinate (MinusB) at (-4,7);
    \draw[->, red, dashed, opacity=0.8] (O) -- (MinusB) node[above left] {$-b$};
    
    % Second, vector \bar{a}
    \coordinate (Abar) at (-6,-5);
    \draw[->, cyan, dashed, opacity=0.8] (O) -- (Abar) node[below left]{$\bar{a}$};
    
    
    % Summing Abar + (-B)
    \coordinate (AbarMinusB) at (-10, 2);
    \draw[dashed, gray] (Abar) -- (AbarMinusB);
    \draw[dashed, gray] (MinusB) -- (AbarMinusB);
    \draw[->, purple] (O) -- (AbarMinusB) node[left] {$\bar{a}-b$};
    
    % Final sum: (Abar - B) + 2C
    \draw[dashed, gray] (AbarMinusB) -- (Z);
    \draw[dashed, gray] (TwoC) -- (Z);
    \draw[->, green!60!black, ultra thick] (O) -- (Z) node[above] {$z = \bar{a}-b+2c$};

    % Points
    \fill[black] (Z) circle (3pt);
\end{tikzpicture}
\end{document}
```
