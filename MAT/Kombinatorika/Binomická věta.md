#MAT #MAT/Kombinatorika 
[[Combination]], [[Pascalův trojúhelník]]
$$
\begin{gathered}
a,b \in R,n\in N~\text{platí} \\
(a+b)^n
=\begin{pmatrix}
n \\
0
\end{pmatrix}a^nb^0+\begin{pmatrix}
n \\
n-1
\end{pmatrix}a^{n-1}b^1+\begin{pmatrix}
n \\
n-2
\end{pmatrix}a^{n-2}b^2+\dots+\begin{pmatrix}
n \\
k
\end{pmatrix}a^{n-k}b^k+\dots+\begin{pmatrix}
n \\
n
\end{pmatrix}a^{n-n}b^n\end{gathered}
$$
