$$
\begin{gathered}
\boxed{ n! = n\cdot(n-1) \cdot(n-2)\cdot ~\dots~ \cdot 1}\\
\boxed{n! = n\cdot(n-1)!}
\end{gathered}
$$
---

$$
\begin{align}
&\frac{n^2-9}{(n+3)!}+\frac{6}{(n+2)!}-\frac{1}{(n+1)!}= \\
=& \frac{(n-3)(n+3)}{(n+3)!}+\frac{6}{(n+2)!}-\frac{1}{(n+1)!}= \\
=& \frac{(n-3)}{(n+2)!}+\frac{6}{(n+2)!}-\frac{1}{(n+1)!}= \\
=& \frac{(n-3)+6-1\cdot(n+2)}{(n+2)!} \\
=& \frac{1}{(n+2)!}
\end{align}
$$
